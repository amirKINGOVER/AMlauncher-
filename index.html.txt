    <script>
        function openModal(modalId) {
            document.getElementById(modalId).style.display = 'flex';
        }

        function closeModal(modalId) {
            document.getElementById(modalId).style.display = 'none';
        }

        function downloadVersion(versionType) {
            const baseURL = "https://raw.githubusercontent.com/amirKINGOVER/AMlauncher-/main/";
            
            if (versionType === '1.2') {
                window.location.href = baseURL + 'AMlauncher.1.2.rar';
            } else {
                window.location.href = baseURL + 'AMlauncher1.0.zip';
            }
            
            closeModal('downloadModal');
        }

        function saveProfile() {
            const username = document.getElementById('gameUsername').value;
            if(!username) {
                alert('الرجاء كتابة اسم المستخدم أولاً!');
                return;
            }
            alert('تم حفظ اسم المستخدم (' + username + ') بنجاح!');
            closeModal('profileModal');
        }
